# usage: live.sh <label>; runs the daemon, waits for "fully operational", SIGTERMs it, prints shutdown log lines
export JWT_SECRET_KEY=throwaway-3134-$(date +%s)-abcdefghijklmnopqrstuvwxyz POSTGRES_DB=shutdown3134 DAEMON_HEALTH_PORT=39191 DAEMON_METRICS_PORT=39190 DAEMON_WEBHOOK_ENABLED=false
LOG=$TMPDIR/daemon-$1.log
